# 类

## 例子

```rust
struct Demo {
    width: u32,
    height: u32,
}

impl Demo {
    pub fn new(width: u32, height: u32) -> Self {
        Self { width, height }
    }

    pub fn aaa(&self) -> u32 {
        self.width * self.height
    }

    pub fn bbb(width: u32, height: u32) -> bool {
        width == height
    }
}

fn main() {
    let bbb = Demo::bbb(10, 10);
    println!("bbb: {}", bbb);

    let rect = Demo::new(30, 50);
    let aaa: u32 = rect.aaa();
    println!("aaa: {}", aaa);
}
```
