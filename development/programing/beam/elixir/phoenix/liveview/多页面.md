# 多页面

## 说明

默认 liveview 只支持单一页面

很多时候，需要多个页面，有的是 liveview, 有的不是

## 例子

router.ex

```elixir
scope "/", WebDemoWeb do
  pipe_through :browser

  live "/page1", Page1Live, :index
  get "/page2", PageController, :page2
end
```

page1_live.ex

```elixir
defmodule WebDemoWeb.Page1Live do
  use WebDemoWeb, :live_view

  alias WebDemo.DemoWorker

  @impl true
  def mount(_params, _session, socket) do
    if connected?(socket) do
      Phoenix.PubSub.subscribe(WebDemo.PubSub, "worker")
    end

    {:ok, assign(socket, count: DemoWorker.get_count())}
  end

  @impl true
  def handle_info({:tick, count}, socket) do
    {:noreply, assign(socket, count: count)}
  end
end
```

page1_live.html.heex

```html
<Layouts.app flash={@flash}>
  <h1>Page 1 (LiveView 长连接)</h1>
  <p>实时计数: <span id="count">{@count}</span></p>
  <.link href={~p"/page2"}>去 Page 2</.link>
</Layouts.app>
```

page_controller.ex

```elixir
defmodule WebDemoWeb.PageController do
  use WebDemoWeb, :controller

  def page2(conn, _params) do
    render(conn, :page2)
  end
end
```

page2.html.heex

```html
<h1>Page 2 (普通页面)</h1>
<p>这是没有任何长连接的普通页面。</p>
<.link href={~p"/page1"}>去 Page 1</.link>
```

demo_worker.ex

```elixir
defmodule WebDemo.DemoWorker do
  use GenServer

  def start_link(_opts) do
    GenServer.start_link(__MODULE__, 0, name: __MODULE__)
  end

  def get_count do
    GenServer.call(__MODULE__, :get)
  end

  @impl true
  def init(count) do
    :timer.send_interval(1000, :tick)
    {:ok, count}
  end

  @impl true
  def handle_call(:get, _from, count) do
    {:reply, count, count}
  end

  @impl true
  def handle_info(:tick, count) do
    new_count = count + 1
    Phoenix.PubSub.broadcast(WebDemo.PubSub, "worker", {:tick, new_count})
    {:noreply, new_count}
  end
end
```
